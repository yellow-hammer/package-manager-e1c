Функция СвойствоОбъекта(Объект, ИмяСвойства)

	Если Объект = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	Если ТипЗнч(Объект) = Тип("Структура") Тогда
		Значение = Неопределено;
		Если Объект.Свойство(ИмяСвойства, Значение) Тогда
			Возврат Значение;
		КонецЕсли;
		Возврат Неопределено;
	КонецЕсли;

	Если ТипЗнч(Объект) = Тип("Соответствие") Тогда
		Возврат Объект[ИмяСвойства];
	КонецЕсли;

	Возврат Неопределено;

	КонецЦикла;

	Возврат Результат;

Функция ЗначениеСвойстваМетаданных(ПутьXml, ИмяСвойства)

	Если Не ФайлСуществует(ПутьXml) Тогда
		Возврат "";
	КонецЕсли;

	Чтение = Новый ЧтениеXML;
	Чтение.ОткрытьФайл(ПутьXml);
	ВнутриProperties = Ложь;
	Значение = "";

	Пока Чтение.Прочитать() Цикл

		Если Чтение.ТипУзла = ТипУзлаXML.НачалоЭлемента Тогда

			Если ЭтоИмяУзла(Чтение, "Properties") Тогда
				ВнутриProperties = Истина;
			ИначеЕсли ВнутриProperties И ЭтоИмяУзла(Чтение, ИмяСвойства) Тогда
				Чтение.Прочитать();
				Если Чтение.ТипУзла = ТипУзлаXML.Текст Тогда
					Значение = СокрЛП(Чтение.Значение);
				КонецЕсли;
				Прервать;
			КонецЕсли;

		ИначеЕсли Чтение.ТипУзла = ТипУзлаXML.КонецЭлемента
			И ЭтоИмяУзла(Чтение, "Properties") Тогда
			Прервать;
		КонецЕсли;

	КонецЦикла;

	Чтение.Закрыть();

	Возврат Значение;

Функция СтрокаВКавычкахПосле(Текст, Маркер)

	Позиция = СтрНайти(Текст, Маркер);
	Если Позиция = 0 Тогда
		Возврат "";
	КонецЕсли;

	Хвост = Сред(Текст, Позиция + СтрДлина(Маркер));
	Начало = СтрНайти(Хвост, """");
	Если Начало = 0 Тогда
		Возврат "";
	КонецЕсли;

	Хвост = Сред(Хвост, Начало + 1);
	КонецКавычек = СтрНайти(Хвост, """");
	Если КонецКавычек = 0 Тогда
		Возврат "";
	КонецЕсли;

	Возврат СокрЛП(Лев(Хвост, КонецКавычек - 1));

Функция КореньGit(КаталогИсходников)

	Текущий = Новый Файл(КаталогИсходников).ПолноеИмя;

	Пока Истина Цикл

		Если ФайлСуществует(ФС.ОбъединитьПути(Текущий, ".git")) Тогда
			Возврат Текущий;
		КонецЕсли;

		Родитель = Новый Файл(Текущий).Путь;
		Если Не ЗначениеЗаполнено(Родитель) Тогда
			Прервать;
		КонецЕсли;

		РодительПолный = Новый Файл(Родитель).ПолноеИмя;
		Если НормализоватьПуть(РодительПолный) = НормализоватьПуть(Текущий) Тогда
			Прервать;
		КонецЕсли;

		Текущий = РодительПолный;

	КонецЦикла;

	ВызватьИсключение "Not a git work tree: " + КаталогИсходников;

Функция ПрефиксВРепозитории(КаталогИсходников, Корень)

	АбсолютныйКаталог = СтрЗаменить(Новый Файл(КаталогИсходников).ПолноеИмя, "\", "/");
	АбсолютныйКорень = СтрЗаменить(Новый Файл(Корень).ПолноеИмя, "\", "/");

	Пока СтрЗаканчиваетсяНа(АбсолютныйКорень, "/") Цикл
		АбсолютныйКорень = Лев(АбсолютныйКорень, СтрДлина(АбсолютныйКорень) - 1);
	КонецЦикла;

	Если НормализоватьПуть(АбсолютныйКаталог) = НормализоватьПуть(АбсолютныйКорень) Тогда
		Возврат "";
	КонецЕсли;

	Если Не СтрНачинаетсяС(НормализоватьПуть(АбсолютныйКаталог), НормализоватьПуть(АбсолютныйКорень) + "/") Тогда
		ВызватьИсключение "Path is outside git work tree: " + КаталогИсходников;
	КонецЕсли;

	Префикс = Сред(АбсолютныйКаталог, СтрДлина(АбсолютныйКорень) + 2);
	Пока СтрНачинаетсяС(Префикс, "/") Цикл
		Префикс = Сред(Префикс, 2);
	КонецЦикла;

	Возврат Префикс;

// Тег - это релиз, поэтому публикуется только тег, коммит которого уже
// в ветке релиза. Проверка здесь, а не в yaml: раннер бывает и Windows,
// и Linux, а синтаксис их оболочек разный.
Процедура ПроверитьТегВеткиРелиза()

	Тег = ПеременнаяСреды("CI_COMMIT_TAG");
	Ветка = ВеткаРелиза();

	Коммит = ПолучитьПеременнуюСреды("CI_COMMIT_SHA");
	Если Не ЗначениеЗаполнено(Коммит) Тогда
		Коммит = "HEAD";
	КонецЕсли;

	Параметры = Новый Массив;
	Параметры.Добавить("fetch");
	Параметры.Добавить("--quiet");
	Параметры.Добавить("origin");
	Параметры.Добавить(Ветка);
	Если КодGit(КаталогПроекта, Параметры) <> 0 Тогда
		ВызватьИсключение "cannot fetch origin/" + Ветка + ". Set GIT_DEPTH to 0 in the job.";
	КонецЕсли;

	Параметры = Новый Массив;
	Параметры.Добавить("merge-base");
	Параметры.Добавить("--is-ancestor");
	Параметры.Добавить(Коммит);
	Параметры.Добавить("origin/" + Ветка);
	Если КодGit(КаталогПроекта, Параметры) <> 0 Тогда
		ВызватьИсключение "tag " + Тег + " is not on " + Ветка + ". Release tags belong to " + Ветка + ".";
	КонецЕсли;

	Сообщить("Tag " + Тег + " is on " + Ветка);

Функция ОтветHttpCurl(Токен, Url, ПутьТела, Метод = "")

	ПутьОтвета = ПолучитьИмяВременногоФайла("body");
	Команда = Новый Команда;
	Команда.УстановитьКоманду("curl.exe");
	Команда.ДобавитьПараметр("--silent");
	Команда.ДобавитьПараметр("--show-error");
	Команда.ДобавитьПараметр("--output");
	Команда.ДобавитьПараметр(ПутьОтвета);
	Команда.ДобавитьПараметр("--write-out");
	Команда.ДобавитьПараметр("%{http_code}");
	Если ЗначениеЗаполнено(Метод) Тогда
		Команда.ДобавитьПараметр("--request");
		Команда.ДобавитьПараметр(Метод);
	КонецЕсли;
	Команда.ДобавитьПараметр("--header");
	Команда.ДобавитьПараметр("PRIVATE-TOKEN:" + Токен);
	Если ЗначениеЗаполнено(ПутьТела) Тогда
		Команда.ДобавитьПараметр("--header");
		Команда.ДобавитьПараметр("Content-Type:application/json");
		Команда.ДобавитьПараметр("--data-binary");
		Команда.ДобавитьПараметр("@" + ПутьТела);
	КонецЕсли;
	Команда.ДобавитьПараметр(Url);
	Команда.ПоказыватьВыводНемедленно(Ложь);
	Команда.Исполнить();

	Код = СокрЛП(ПоследняяНепустаяСтрока(Команда.ПолучитьВывод()));
	Если Не ЭтоКодHttp(Код) Тогда
		ВызватьИсключение "curl failed: " + Код;
	КонецЕсли;

	Возврат Новый Структура("Код,Тело", Код, ПрочитатьТекст(ПутьОтвета));

Функция ИменаСсылокРелиза(Тело)

	Имена = Новый Соответствие;
	Данные = ПрочитатьJsonИзСтроки(Тело);
	Активы = СвойствоОбъекта(Данные, "assets");
	Список = СвойствоОбъекта(Активы, "links");
	Если ТипЗнч(Список) <> Тип("Массив") Тогда
		Возврат Имена;
	КонецЕсли;

	Для Каждого Элемент Из Список Цикл
		Имя = СвойствоОбъекта(Элемент, "name");
		Если ЗначениеЗаполнено(Имя) Тогда
			Имена.Вставить(Строка(Имя), Истина);
		КонецЕсли;
	КонецЦикла;

	Возврат Имена;

Процедура ДобавитьСсылкиРелиза(Токен, UrlРелиза, Ссылки, УжеЕсть)

	Для Каждого Ссылка Из Ссылки Цикл

		Если УжеЕсть.Получить(Ссылка.Имя) = Истина Тогда
			Сообщить("link exists " + Ссылка.Имя);
			Продолжить;
		КонецЕсли;

		ПутьТела = ФС.ОбъединитьПути(КаталогВыхода, "release-link.json");
		ЗаписатьТелоСсылкиРелиза(ПутьТела, Ссылка);
		Ответ = ОтветHttpCurl(Токен, UrlРелиза + "/assets/links", ПутьТела);
		Если Ответ.Код <> "201" Тогда
			ВызватьИсключение "Release link failed: HTTP " + Ответ.Код + " " + Ответ.Тело;
		КонецЕсли;
		Сообщить("link added " + Ссылка.Имя);

Процедура ОбеспечитьРелиз()

	Тег = ПеременнаяСреды("CI_COMMIT_TAG");
	Токен = ТокенРелиза();
	База = СтрШаблон("%1/projects/%2/releases",
		ПеременнаяСреды("CI_API_V4_URL"),
		ПеременнаяСреды("CI_PROJECT_ID"));
	UrlРелиза = База + "/" + КодироватьСегментURL(Тег);
	Ссылки = СсылкиПакетовРелиза(Токен);
	Описание = ОписаниеРелиза(Ссылки);

	Ответ = ОтветHttpCurl(Токен, UrlРелиза, "");
	Если Ответ.Код = "200" Тогда
		ДобавитьСсылкиРелиза(Токен, UrlРелиза, Ссылки, ИменаСсылокРелиза(Ответ.Тело));
		ОбновитьОписаниеРелиза(Токен, UrlРелиза, Описание);
		Сообщить("release " + Тег + " updated");
		Возврат;
	КонецЕсли;
	Если Ответ.Код = "401" Или Ответ.Код = "403" Тогда
		ВызватьИсключение "Release token cannot access Releases (HTTP " + Ответ.Код
			+ "). Check RELEASE_TOKEN / GROUP_ACCESS_TOKEN scope and Protected flag.";
	КонецЕсли;
	Если Ответ.Код <> "404" Тогда
		ВызватьИсключение "Release lookup failed: HTTP " + Ответ.Код + " " + Ответ.Тело;
	КонецЕсли;

	КаталогВыхода = ФС.ОбъединитьПути(КаталогПроекта, "build/out");
	ФС.ОбеспечитьКаталог(КаталогВыхода);
	ПутьТела = ФС.ОбъединитьПути(КаталогВыхода, "release.json");
	ЗаписатьТелоРелиза(ПутьТела, Тег, Описание, Ссылки);

	Ответ = ОтветHttpCurl(Токен, База, ПутьТела);
	Если Ответ.Код = "201" Тогда
		Сообщить("release " + Тег + " created");
		Возврат;
	КонецЕсли;
	Если Ответ.Код = "409" Тогда
		ДобавитьСсылкиРелиза(Токен, UrlРелиза, Ссылки, Новый Соответствие);
		ОбновитьОписаниеРелиза(Токен, UrlРелиза, Описание);
		Сообщить("release " + Тег + " updated");
		Возврат;
	КонецЕсли;

	ВызватьИсключение "Release create failed: HTTP " + Ответ.Код + " " + Ответ.Тело;

КонецПроцедуры

Процедура ОпубликоватьПакеты()

	ОбеспечитьБазуДляВнешнихФайлов();

	Для Каждого Пакет Из ПрочитатьМанифест() Цикл

		ОтносительныйПуть = СвойствоОбъекта(Пакет, "path");
		Если ОтносительныйПуть = Неопределено Или Не ЗначениеЗаполнено(СокрЛП(Строка(ОтносительныйПуть))) Тогда
			ВызватьИсключение "pm-packages.json: package without path";
		КонецЕсли;

		КаталогИсходников = СокрЛП(Строка(ОтносительныйПуть));
		Если Не ФайлСуществует(КаталогИсходников) Тогда
			ВызватьИсключение "Path from pm-packages.json not found: " + КаталогИсходников;
		КонецЕсли;

		ОпубликоватьПакет(ОпределитьВид(Пакет, КаталогИсходников), КаталогИсходников);

	КонецЦикла;
