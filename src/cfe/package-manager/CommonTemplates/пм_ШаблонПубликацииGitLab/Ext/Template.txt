# Сборка .cfe и публикация в GitLab Generic Package Registry.
# Плейсхолдеры @...@ подставляет обработка «Генератор шаблонов публикации».
#
# Имя package берется из Name в Configuration.xml. GitLab Generic
# принимает только ASCII: не-ASCII Name транслитерируется в ключ
# реестра и имя файла. Внутри .cfe Name не меняется. Списки алиасов
# в пайплайне не заводим.
# Каталог исходников ищется в src/cfe по Name.
#
# Если в репозитории уже есть .gitlab-ci.yml, не затирайте его:
# сохраните этот файл отдельно и подключите через include.
# Теги раннера задайте сами (на корпоративном GitLab 1С обычно windows, ones).

stages:
  - build
  - package

variables:
  PACKAGE_NAME: "@PACKAGE_NAME@"
  CFE_NAME: "@CFE_NAME@"
  CFE_PATH: "@CFE_PATH@"

build:cfe:
  stage: build
  rules:
    - if: $CI_COMMIT_TAG
  script:
    - |
      $ErrorActionPreference = "Stop"
      function Get-ExtensionName([string]$srcDir) {
        $cfg = Join-Path $srcDir "Configuration.xml"
        if (-not (Test-Path $cfg)) {
          throw "No Configuration.xml in $srcDir"
        }
        $xml = New-Object System.Xml.XmlDocument
        $xml.Load((Resolve-Path $cfg))
        $nsmgr = New-Object System.Xml.XmlNamespaceManager($xml.NameTable)
        $nsmgr.AddNamespace("md", "http://v8.1c.ru/8.3/MDClasses")
        $node = $xml.SelectSingleNode("/md:MetaDataObject/md:Configuration/md:Properties/md:Name", $nsmgr)
        if (-not $node -or [string]::IsNullOrWhiteSpace($node.InnerText)) {
          throw "Name not found in $cfg"
        }
        return $node.InnerText.Trim()
      }

      $src = $null
      Get-ChildItem -Path "src/cfe" -Directory | ForEach-Object {
        if ((Get-ExtensionName $_.FullName) -eq $env:PACKAGE_NAME) {
          $src = $_.FullName
        }
      }
      if (-not $src) {
        throw "No extension with Name=$env:PACKAGE_NAME in src/cfe"
      }

      New-Item -ItemType Directory -Force -Path "build/out/cfe" | Out-Null
      vrunner compileexttocfe --settings env.json --src $src --out "$env:CFE_PATH"
      if ($LASTEXITCODE -ne 0) {
        throw "compileexttocfe failed for $env:PACKAGE_NAME"
      }
  artifacts:
    paths:
      - "${CFE_PATH}"
    expire_in: 1 day

package:gitlab:
  stage: package
  rules:
    - if: $CI_COMMIT_TAG
  needs:
    - job: build:cfe
      artifacts: true
  script:
    - |
      $ErrorActionPreference = "Stop"
      function ConvertTo-GitLabPackageName([string]$name) {
        if ($name -match '^[A-Za-z0-9._~@+-]+$') {
          return $name
        }
        $map = @{
          0x0410='A'; 0x0430='a'; 0x0411='B'; 0x0431='b'
          0x0412='V'; 0x0432='v'; 0x0413='G'; 0x0433='g'
          0x0414='D'; 0x0434='d'; 0x0415='E'; 0x0435='e'
          0x0401='Yo'; 0x0451='yo'; 0x0416='Zh'; 0x0436='zh'
          0x0417='Z'; 0x0437='z'; 0x0418='I'; 0x0438='i'
          0x0419='Y'; 0x0439='y'; 0x041A='K'; 0x043A='k'
          0x041B='L'; 0x043B='l'; 0x041C='M'; 0x043C='m'
          0x041D='N'; 0x043D='n'; 0x041E='O'; 0x043E='o'
          0x041F='P'; 0x043F='p'; 0x0420='R'; 0x0440='r'
          0x0421='S'; 0x0441='s'; 0x0422='T'; 0x0442='t'
          0x0423='U'; 0x0443='u'; 0x0424='F'; 0x0444='f'
          0x0425='H'; 0x0445='h'; 0x0426='Ts'; 0x0446='ts'
          0x0427='Ch'; 0x0447='ch'; 0x0428='Sh'; 0x0448='sh'
          0x0429='Sch'; 0x0449='sch'; 0x042B='Y'; 0x044B='y'
          0x042D='E'; 0x044D='e'; 0x042E='Yu'; 0x044E='yu'
          0x042F='Ya'; 0x044F='ya'
        }
        $parts = New-Object System.Collections.Generic.List[string]
        foreach ($ch in $name.ToCharArray()) {
          $code = [int]$ch
          if ($map.ContainsKey($code)) {
            $parts.Add([string]$map[$code])
          } elseif ([string]$ch -match '^[A-Za-z0-9._~@+-]$') {
            $parts.Add([string]$ch)
          }
        }
        $result = -join $parts.ToArray()
        if ($result -notmatch '^[A-Za-z0-9._~@+-]+$') {
          throw "GitLab Generic package name must be ASCII. Cannot derive one from Name"
        }
        return $result
      }

      $packageName = ConvertTo-GitLabPackageName $env:PACKAGE_NAME
      $cfeName = "$packageName.cfe"
      $package = [uri]::EscapeDataString($packageName)
      $version = [uri]::EscapeDataString($env:CI_COMMIT_TAG)
      $file = [uri]::EscapeDataString($cfeName)
      $url = "$env:CI_API_V4_URL/projects/$env:CI_PROJECT_ID/packages/generic/$package/$version/$file"
      curl.exe --fail --header "JOB-TOKEN: $env:CI_JOB_TOKEN" --upload-file "$env:CFE_PATH" $url
      if ($LASTEXITCODE -ne 0) {
        throw "Package upload failed for $cfeName"
      }
